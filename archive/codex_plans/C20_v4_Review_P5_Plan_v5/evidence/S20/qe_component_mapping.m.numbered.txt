   1 | function m = qe_component_mapping(p,lb,ub,n,scale,ampunit,peak_model)
   2 | % Optimizer slots -> energy-ordered labels, with original v2 tolerance.
   3 | p=p(:).'; lb=lb(:).'; ub=ub(:).';
   4 | assert(numel(p)==numel(lb)&&numel(p)==numel(ub));
   5 | pars=reshape(p(3:2+3*n),3,n).';
   6 | [~,order]=sort(pars(:,1)); inverse=zeros(1,n); inverse(order)=1:n;
   7 | native_scale=[scale 1 repmat([1000 1000 ampunit*scale],1,n)];
   8 | names=["background_B","background_r",repmat(["E0","width","A"],1,n)];
   9 | units=["counts","dimensionless",repmat(["meV","meV","counts*meV"],1,n)];
  10 | if strcmp(peak_model,'lorentz'), units(5:3:2+3*n)="counts*meV^2"; end
  11 | ordered=zeros(size(p)); raw_component=zeros(size(p));
  12 | for j=1:n
  13 |  ii=2+3*(j-1)+(1:3); ordered(ii)=inverse(j); raw_component(ii)=j;
  14 | end
  15 | if numel(p)>2+3*n
  16 |  native_scale(end+1)=scale; names(end+1)="background_C"; units(end+1)="counts";
  17 | end
  18 | tol_lo=1e-5*max(1,abs(lb)); tol_hi=1e-5*max(1,abs(ub));
  19 | tol_lo(~isfinite(ub))=1e-5;
  20 | lower=isfinite(p)&abs(p-lb)<tol_lo;
  21 | upper=isfinite(p)&isfinite(ub)&abs(p-ub)<tol_hi;
  22 | types=repmat("interior",size(p));
  23 | types(lower)=names(lower)+"_lower"; types(upper)=names(upper)+"_upper";
  24 | types(~isfinite(p))="not_assessable";
  25 | m=struct('order',order,'raw_to_order',inverse,'raw_component',raw_component, ...
  26 |  'ordered_component',ordered,'parameters',pars(order,:).*[1000 1000 ampunit*scale], ...
  27 |  'native_scale',native_scale,'names',names,'units',units,'lower_hit',lower,'upper_hit',upper, ...
  28 |  'tol_lower',tol_lo,'tol_upper',tol_hi,'boundary_type',types,'finite',all(isfinite(p)), ...
  29 |  'equal_center',n==2 && abs(pars(1,1)-pars(end,1))<1e-8);
  30 | end
